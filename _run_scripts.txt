# -----------------------------
# Argus: ONE-RUN nightly (git + backup + lab backtest + lab live + prod backtest + prod live)
# Run this from PowerShell.
# -----------------------------

# 0) Repo root
Set-Location C:\Argus\repo

# 1) Git: commit if needed + push
git status
git add -A
git diff --cached --quiet
if ($LASTEXITCODE -ne 0) {
    git commit -m "nightly: fixes"
}
git push

# 2) Backup + verify (zip proof of current state)
powershell.exe -ExecutionPolicy Bypass -File C:\Argus\ops\backup_repo.ps1
powershell.exe -ExecutionPolicy Bypass -File C:\Argus\ops\backup_verify.ps1

# 3) LAB: backtest
$env:ARGUS_ARTIFACT_ROOT = "C:\Argus\argus-lab"
cmd.exe /c C:\Argus\ops\run_backtest.cmd
Get-ChildItem C:\Argus\argus-lab\ops\logs\bt_summary_*.json |
  Sort-Object LastWriteTime -Descending |
  Select-Object -First 1 Name, LastWriteTime

# 4) LAB: live (paper) in a new window
Start-Process -FilePath "cmd.exe" -ArgumentList "/c C:\Argus\ops\run_live.cmd" -WindowStyle Normal

# 5) PROD: backtest
$env:ARGUS_ARTIFACT_ROOT = "C:\Argus\argus-prod"
cmd.exe /c C:\Argus\ops\run_backtest.cmd
Get-ChildItem C:\Argus\argus-prod\ops\logs\bt_summary_*.json |
  Sort-Object LastWriteTime -Descending |
  Select-Object -First 1 Name, LastWriteTime

# 6) PROD: live in a new window
Start-Process -FilePath "cmd.exe" -ArgumentList "/c C:\Argus\ops\run_live.cmd" -WindowStyle Normal

# 7) Final hygiene: repo must remain clean (no runtime junk)
Set-Location C:\Argus\repo
git status
Get-ChildItem . -Filter *.log -ErrorAction SilentlyContinue
